"""
Pascal Terminal v2 — Backtesting Engine
Tracks past recommendations and scores their accuracy.
Stored in backtest_history.json, updated every week.
"""

import json
import os
import datetime
import requests
from typing import Optional


HISTORY_FILE = os.path.join(os.path.dirname(__file__), "..", "data", "backtest_history.json")


def load_history() -> dict:
    """Load backtest history from disk."""
    if os.path.exists(HISTORY_FILE):
        with open(HISTORY_FILE, encoding="utf-8") as f:
            return json.load(f)
    return {"recommendations": [], "performance_summary": {}}


def save_history(history: dict):
    """Save backtest history to disk."""
    os.makedirs(os.path.dirname(HISTORY_FILE), exist_ok=True)
    with open(HISTORY_FILE, "w", encoding="utf-8") as f:
        json.dump(history, f, ensure_ascii=False, indent=2)


def fetch_price(ticker: str) -> Optional[float]:
    """Fetch current price for a ticker."""
    try:
        url = f"https://query1.finance.yahoo.com/v8/finance/chart/{ticker}?interval=1d&range=5d"
        headers = {"User-Agent": "Mozilla/5.0"}
        r = requests.get(url, headers=headers, timeout=10)
        data = r.json()
        meta = data["chart"]["result"][0]["meta"]
        return meta.get("regularMarketPrice")
    except Exception:
        return None


def record_recommendations(analysis: dict, screened: list, week: int, date: str):
    """
    Record this week's recommendations for future backtesting.
    Called after each briefing generation.
    """
    history = load_history()

    new_rec = {
        "week": week,
        "date": date,
        "recommendations": [],
        "status": "OPEN",  # OPEN / EVALUATED
    }

    # Record portfolio actions from Claude analysis
    for act in analysis.get("actions_portefeuille", []):
        if act.get("action") in ("ACHETER", "RENFORCER"):
            ticker = act.get("ticker", "")
            if ticker:
                new_rec["recommendations"].append({
                    "type":          "PORTEFEUILLE",
                    "ticker":        ticker,
                    "name":          act.get("name", ""),
                    "action":        act.get("action"),
                    "horizon":       act.get("horizon", "CT"),
                    "prix_entree":   act.get("prix_actuel"),
                    "prix_cible":    act.get("prix_cible"),
                    "stop_loss":     act.get("stop_loss"),
                    "conviction":    act.get("conviction", 3),
                    "date_rec":      date,
                    "prix_actuel":   None,  # filled at evaluation
                    "performance":   None,
                    "hit":           None,  # True/False/None
                    "evaluated_at":  None,
                })

    # Record external opportunities
    for opp in analysis.get("opportunites_externes", []):
        if opp.get("action") in ("ACHETER",):
            ticker = opp.get("ticker", "")
            if ticker:
                new_rec["recommendations"].append({
                    "type":         "OPPORTUNITE",
                    "ticker":       ticker,
                    "name":         opp.get("name", ""),
                    "action":       opp.get("action"),
                    "horizon":      opp.get("horizon", "CT"),
                    "theme":        opp.get("theme", ""),
                    "prix_entree":  opp.get("prix_actuel"),
                    "prix_cible":   opp.get("prix_cible"),
                    "stop_loss":    opp.get("stop_loss"),
                    "conviction":   opp.get("conviction", 3),
                    "date_rec":     date,
                    "prix_actuel":  None,
                    "performance":  None,
                    "hit":          None,
                    "evaluated_at": None,
                })

    history["recommendations"].append(new_rec)
    save_history(history)
    print(f"Enregistré {len(new_rec['recommendations'])} recommandations pour la semaine {week}")


def evaluate_past_recommendations():
    """
    Update past recommendations with current prices.
    CT = evaluate after 2 weeks, MT after 8 weeks, LT after 52 weeks.
    """
    history = load_history()
    today = datetime.date.today()
    updated = 0

    for week_batch in history["recommendations"]:
        if week_batch["status"] == "FULLY_EVALUATED":
            continue

        rec_date = datetime.date.fromisoformat(week_batch["date"])
        days_elapsed = (today - rec_date).days

        for rec in week_batch["recommendations"]:
            if rec["hit"] is not None:
                continue  # already evaluated

            horizon = rec.get("horizon", "CT")
            eval_days = {"CT": 14, "MT": 56, "LT": 365}.get(horizon, 14)

            if days_elapsed < eval_days:
                continue  # too early

            # Fetch current price
            current_price = fetch_price(rec["ticker"])
            if not current_price or not rec.get("prix_entree"):
                continue

            entry = rec["prix_entree"]
            target = rec.get("prix_cible")
            stop   = rec.get("stop_loss")

            perf = (current_price - entry) / entry * 100
            rec["prix_actuel"]  = round(current_price, 2)
            rec["performance"]  = round(perf, 2)
            rec["evaluated_at"] = today.isoformat()

            # Determine hit/miss
            if rec["action"] in ("ACHETER", "RENFORCER"):
                if stop and current_price < stop:
                    rec["hit"] = False  # Stop hit
                    rec["result"] = "STOP_HIT"
                elif target and current_price >= target:
                    rec["hit"] = True   # Target hit
                    rec["result"] = "TARGET_HIT"
                elif perf > 5:
                    rec["hit"] = True
                    rec["result"] = f"IN_PROFIT_{perf:.1f}%"
                elif perf < -10:
                    rec["hit"] = False
                    rec["result"] = f"LOSS_{perf:.1f}%"
                else:
                    rec["hit"] = None  # inconclusive
                    rec["result"] = f"OPEN_{perf:.1f}%"
            elif rec["action"] == "SHORT_CANDIDAT":
                if perf < -5:
                    rec["hit"] = True
                    rec["result"] = f"SHORT_WIN_{abs(perf):.1f}%"
                elif perf > 10:
                    rec["hit"] = False
                    rec["result"] = f"SHORT_LOSS_{perf:.1f}%"

            updated += 1

    # Compute summary stats
    all_recs = [r for wb in history["recommendations"] for r in wb["recommendations"]]
    evaluated = [r for r in all_recs if r["hit"] is not None]
    hits       = [r for r in evaluated if r["hit"] is True]
    misses     = [r for r in evaluated if r["hit"] is False]

    by_conviction = {}
    for r in evaluated:
        c = r.get("conviction", 3)
        if c not in by_conviction:
            by_conviction[c] = {"hits": 0, "total": 0}
        by_conviction[c]["total"] += 1
        if r["hit"]:
            by_conviction[c]["hits"] += 1

    by_horizon = {}
    for r in evaluated:
        h = r.get("horizon", "CT")
        if h not in by_horizon:
            by_horizon[h] = {"hits": 0, "total": 0, "avg_perf": []}
        by_horizon[h]["total"] += 1
        if r["hit"]:
            by_horizon[h]["hits"] += 1
        if r.get("performance"):
            by_horizon[h]["avg_perf"].append(r["performance"])

    avg_perfs = {
        h: round(sum(v["avg_perf"]) / len(v["avg_perf"]), 1)
        if v["avg_perf"] else 0
        for h, v in by_horizon.items()
    }

    history["performance_summary"] = {
        "last_updated":       today.isoformat(),
        "total_recommendations": len(all_recs),
        "evaluated":          len(evaluated),
        "hit_rate":           round(len(hits) / len(evaluated) * 100, 1) if evaluated else 0,
        "miss_rate":          round(len(misses) / len(evaluated) * 100, 1) if evaluated else 0,
        "by_conviction":      {
            str(k): {
                "hit_rate": round(v["hits"]/v["total"]*100, 1) if v["total"] else 0,
                "total":    v["total"],
            }
            for k, v in by_conviction.items()
        },
        "by_horizon":         {
            h: {
                "hit_rate": round(v["hits"]/v["total"]*100, 1) if v["total"] else 0,
                "avg_perf": avg_perfs.get(h, 0),
                "total":    v["total"],
            }
            for h, v in by_horizon.items()
        },
        "best_calls": sorted(
            [r for r in evaluated if r.get("performance") and r["hit"]],
            key=lambda x: x["performance"], reverse=True
        )[:5],
        "worst_calls": sorted(
            [r for r in evaluated if r.get("performance") and not r["hit"]],
            key=lambda x: x["performance"]
        )[:5],
    }

    save_history(history)
    print(f"Backtesting: {updated} recommandations évaluées")
    print(f"Hit rate global: {history['performance_summary'].get('hit_rate', 'N/A')}%")
    return history["performance_summary"]


def get_backtest_summary() -> dict:
    """Return the performance summary for dashboard display."""
    history = load_history()
    return history.get("performance_summary", {})


def get_recent_calls(n: int = 10) -> list:
    """Return the most recent recommendations with their outcomes."""
    history = load_history()
    all_recs = []
    for wb in reversed(history["recommendations"]):
        for r in wb["recommendations"]:
            all_recs.append({**r, "week": wb["week"], "batch_date": wb["date"]})
    return all_recs[:n]


if __name__ == "__main__":
    summary = evaluate_past_recommendations()
    print(json.dumps(summary, indent=2, ensure_ascii=False))
